# Native Windows mouse-wheel event.
# One WHEEL_DELTA is one standard mouse-wheel notch.
MOUSEEVENTF_WHEEL = 0x0800
WHEEL_DELTA = 120

# ============================================================
# NATIVE WINDOWS INPUT ENGINE
# ============================================================
# PyAutoGUI is intentionally not used for SCROLL/ZOOM input.
# Windows SendInput gives us low-latency native wheel + keyboard events.
INPUT_MOUSE = 0
INPUT_KEYBOARD = 1

KEYEVENTF_KEYUP = 0x0002

VK_CONTROL = 0x11
VK_SHIFT = 0x10
VK_OEM_PLUS = 0xBB
VK_OEM_MINUS = 0xBD

class MOUSEINPUT(ctypes.Structure):
    _fields_ = [
        ("dx", ctypes.c_long),
        ("dy", ctypes.c_long),
        ("mouseData", ctypes.c_ulong),
        ("dwFlags", ctypes.c_ulong),
        ("time", ctypes.c_ulong),
        ("dwExtraInfo", ctypes.POINTER(ctypes.c_ulong)),
    ]

class KEYBDINPUT(ctypes.Structure):
    _fields_ = [
        ("wVk", ctypes.c_ushort),
        ("wScan", ctypes.c_ushort),
        ("dwFlags", ctypes.c_ulong),
        ("time", ctypes.c_ulong),
        ("dwExtraInfo", ctypes.POINTER(ctypes.c_ulong)),
    ]

class INPUT_UNION(ctypes.Union):
    _fields_ = [
        ("mi", MOUSEINPUT),
        ("ki", KEYBDINPUT),
    ]

class INPUT(ctypes.Structure):
    _anonymous_ = ("u",)
    _fields_ = [
        ("type", ctypes.c_ulong),
        ("u", INPUT_UNION),
    ]

_user32 = ctypes.windll.user32

def native_mouse_wheel(notches: int):
    if notches == 0:
        return
    inp = INPUT()
    inp.type = INPUT_MOUSE
    inp.mi = MOUSEINPUT(
        0, 0,
        ctypes.c_ulong(int(notches) * WHEEL_DELTA),
        0x0800,  # MOUSEEVENTF_WHEEL
        0,
        None,
    )
    _user32.SendInput(1, ctypes.byref(inp), ctypes.sizeof(INPUT))

def _native_key(vk, key_up=False):
    inp = INPUT()
    inp.type = INPUT_KEYBOARD
    inp.ki = KEYBDINPUT(
        int(vk), 0,
        KEYEVENTF_KEYUP if key_up else 0,
        0,
        None,
    )
    _user32.SendInput(1, ctypes.byref(inp), ctypes.sizeof(INPUT))

def native_zoom_in():
    # Ctrl + '+' = Ctrl + Shift + '=' on a standard US-style keyboard.
    _native_key(VK_CONTROL)
    _native_key(VK_SHIFT)
    _native_key(VK_OEM_PLUS)
    _native_key(VK_OEM_PLUS, True)
    _native_key(VK_SHIFT, True)
    _native_key(VK_CONTROL, True)

def native_zoom_out():
    # Ctrl + '-'
    _native_key(VK_CONTROL)
    _native_key(VK_OEM_MINUS)
    _native_key(VK_OEM_MINUS, True)
    _native_key(VK_CONTROL, True)


def distance_2d(p1, p2):
    """Euclidean distance between two normalized 2D landmark points."""
    dx = float(p1[0]) - float(p2[0])
    dy = float(p1[1]) - float(p2[1])
    return (dx * dx + dy * dy) ** 0.5

def send_mouse_wheel(notches):
    """Send a real Windows mouse-wheel event."""
    if not notches:
        return
    ctypes.windll.user32.mouse_event(
        MOUSEEVENTF_WHEEL,
        0,
        0,
        int(notches * WHEEL_DELTA),
        0,
    )

# ONE EURO FILTER
# ============================================================
def smoothing_factor(dt, cutoff):
    r = 2.0 * np.pi * cutoff * dt
    return r / (r + 1.0)

class OneEuroFilter:
    def __init__(self, t0, x0, min_cutoff=0.55, beta=0.35, d_cutoff=1.0):
        self.min_cutoff = float(min_cutoff)
        self.beta = float(beta)
        self.d_cutoff = float(d_cutoff)
        self.x_prev = float(x0)
        self.dx_prev = 0.0
        self.t_prev = float(t0)

    def reset(self, t, x):
        self.x_prev = float(x)
        self.dx_prev = 0.0
        self.t_prev = float(t)

    def __call__(self, t, x):
        dt = float(t - self.t_prev)
        if dt <= 0.000001:
            return self.x_prev
        dt = min(dt, 0.100)

        dx = (float(x) - self.x_prev) / dt
        a_d = smoothing_factor(dt, self.d_cutoff)
        dx_hat = a_d * dx + (1.0 - a_d) * self.dx_prev

        cutoff = self.min_cutoff + self.beta * abs(dx_hat)
        a = smoothing_factor(dt, cutoff)
        x_hat = a * float(x) + (1.0 - a) * self.x_prev

        self.x_prev = x_hat
        self.dx_prev = dx_hat
        self.t_prev = float(t)
        return x_hat

# ROBUST 3-SAMPLE MEDIAN
# ============================================================
class Median3:
    def __init__(self, initial):
        self.values = deque(maxlen=3)
        self.values.append(float(initial))

    def reset(self, value):
        self.values.clear()
        self.values.append(float(value))

    def update(self, value):
        self.values.append(float(value))
        return float(np.median(np.asarray(self.values, dtype=np.float64)))

# ============================================================
# STATIONARY LOCK WITH HYSTERESIS
# ============================================================
class StationaryLock:
    def __init__(self, enter_speed_px=7.0, exit_speed_px=14.0, enter_frames=7, exit_frames=2):
        self.enter_speed_px = float(enter_speed_px)
        self.exit_speed_px = float(exit_speed_px)
        self.enter_frames = int(enter_frames)
        self.exit_frames = int(exit_frames)
        self.locked = False
        self.still_count = 0
        self.move_count = 0

    def update(self, speed_px_s, dt):
        displacement = speed_px_s * dt
        if self.locked:
            if displacement >= self.exit_speed_px * dt:
                self.move_count += 1
            else:
                self.move_count = 0
            if self.move_count >= self.exit_frames:
                self.locked = False
                self.still_count = 0
                self.move_count = 0
        else:
            if displacement <= self.enter_speed_px * dt:
                self.still_count += 1
            else:
                self.still_count = 0
            if self.still_count >= self.enter_frames:
                self.locked = True
                self.move_count = 0
        return self.locked

# ============================================================
class StableBool:
    def __init__(self, on_frames=3, off_frames=4):
        self.on_frames = int(on_frames)
        self.off_frames = int(off_frames)
        self.state = False
        self.on_count = 0
        self.off_count = 0

    def update(self, value):
        value = bool(value)
        if value:
            self.on_count += 1
            self.off_count = 0
            if not self.state and self.on_count >= self.on_frames:
                self.state = True
        else:
            self.off_count += 1
            self.on_count = 0
            if self.state and self.off_count >= self.off_frames:
                self.state = False
        return self.state

def palm_center(landmarks):
    ids = [0, 5, 9, 13, 17]
    pts = np.array([[landmarks[i].x, landmarks[i].y] for i in ids], dtype=np.float64)
    return np.mean(pts, axis=0)

def normalized_pinch_distance(landmarks):
    thumb = np.array([landmarks[4].x, landmarks[4].y], dtype=np.float64)
    index = np.array([landmarks[8].x, landmarks[8].y], dtype=np.float64)
    wrist = np.array([landmarks[0].x, landmarks[0].y], dtype=np.float64)
    middle_mcp = np.array([landmarks[9].x, landmarks[9].y], dtype=np.float64)

    pinch_distance = float(np.linalg.norm(thumb - index))
    palm_scale = max(float(np.linalg.norm(middle_mcp - wrist)), 1e-4)

    # Normalize by apparent palm size so pinch sensitivity stays consistent
    # when the hand moves closer to or farther from the camera.
    return pinch_distance / palm_scale

def reset_pinch_state():
    global pinch_active, pinch_distance_ema, pinch_reference_distance
    global pinch_start_count, pinch_release_count, pinch_accumulator
    pinch_active = False
    pinch_distance_ema = None
    pinch_reference_distance = None
    pinch_start_count = 0
    pinch_release_count = 0
    pinch_accumulator = 0.0

def update_pinch_zoom(landmarks, now):
    """
    Returns:
        +1 = zoom IN
        -1 = zoom OUT
         0 = no zoom

    Pinch distance is normalized by palm size so camera distance does not
    dramatically change the response.
    """
    global pinch_active
    global pinch_distance_ema, pinch_reference_distance
    global pinch_start_count, pinch_release_count
    global pinch_last_zoom_time, pinch_accumulator

    distance = normalized_pinch_distance(landmarks)

    # Keep the EMA alive even when the hand is temporarily outside the
    # comfortable pinch range.
    if not (PINCH_MIN_DISTANCE <= distance <= PINCH_MAX_DISTANCE):
        return 0

    if pinch_distance_ema is None:
        pinch_distance_ema = distance
    else:
        pinch_distance_ema = (
            PINCH_EMA_ALPHA * distance
            + (1.0 - PINCH_EMA_ALPHA) * pinch_distance_ema
        )

    d = pinch_distance_ema

    # ------------------------------------------------------------
    # PINCH START
    # ------------------------------------------------------------
    if not pinch_active:
        if d <= PINCH_START_DISTANCE:
            pinch_start_count += 1
        else:
            pinch_start_count = 0

        if pinch_start_count >= PINCH_START_FRAMES:
            pinch_active = True
            pinch_reference_distance = d
            pinch_accumulator = 0.0
            pinch_release_count = 0

        return 0

    # ------------------------------------------------------------
    # PINCH RELEASE
    # ------------------------------------------------------------
    if d >= PINCH_RELEASE_DISTANCE:
        pinch_release_count += 1
    else:
        pinch_release_count = 0

    if pinch_release_count >= PINCH_RELEASE_FRAMES:
        pinch_active = False
        pinch_reference_distance = None
        pinch_accumulator = 0.0
        pinch_start_count = 0
        pinch_release_count = 0
        return 0

    if pinch_reference_distance is None:
        pinch_reference_distance = d
        return 0

    # ------------------------------------------------------------
    # RELATIVE PINCH MOVEMENT
    #
    # d gets SMALLER  -> thumb/index move together -> ZOOM IN
    # d gets LARGER   -> thumb/index move apart    -> ZOOM OUT
    # ------------------------------------------------------------
    delta = pinch_reference_distance - d
    pinch_accumulator += delta
    pinch_reference_distance = d

    if now - pinch_last_zoom_time < PINCH_ZOOM_COOLDOWN:
        return 0

    if pinch_accumulator >= PINCH_ZOOM_STEP:
        pinch_accumulator -= PINCH_ZOOM_STEP
        pinch_last_zoom_time = now
        return +1

    if pinch_accumulator <= -PINCH_ZOOM_STEP:
        pinch_accumulator += PINCH_ZOOM_STEP
        pinch_last_zoom_time = now
        return -1

    return 0

                relative_y = cursor_tip_y - cursor_anchor_y

                # stationary. When movement resumes, displacement is measured

                    locked_hand_y = float(cursor_tip_y)

                    float(np.linalg.norm(middle_mcp_pt - wrist_pt)),

                    ADAPTIVE_CURSOR_MIN_GAIN,
                    ADAPTIVE_CURSOR_MAX_GAIN

                    ADAPTIVE_CURSOR_DEADZONE_MAX / adaptive_gain,
                    ADAPTIVE_CURSOR_DEADZONE_MIN,
                    ADAPTIVE_CURSOR_DEADZONE_MAX
                ))

                locked_relative_x = (cursor_tip_x - locked_hand_x) * adaptive_gain
                locked_relative_y = (cursor_tip_y - locked_hand_y) * adaptive_gain

                target_x = current_cursor_x + (
                    locked_relative_x * usable_width * CURSOR_SENSITIVITY
                )
                target_y = current_cursor_y + (
                    locked_relative_y * usable_height * CURSOR_SENSITIVITY
                )

                target_x = float(target_x)
                target_y = float(target_y)

                # Final screen-space smoothing removes the small residual
                # oscillation that gets amplified by cursor sensitivity.
                if cursor_screen_x_filter is None:
                    cursor_screen_x_filter = OneEuroFilter(
                        now, target_x, min_cutoff=0.72, beta=0.28, d_cutoff=1.0
                    )
                    cursor_screen_y_filter = OneEuroFilter(
                        now, target_y, min_cutoff=0.72, beta=0.28, d_cutoff=1.0
                    )

                target_x = cursor_screen_x_filter(now, target_x)
                target_y = cursor_screen_y_filter(now, target_y)

                dx_cursor = target_x - current_cursor_x
                dy_cursor = target_y - current_cursor_y
                displacement = float(np.hypot(dx_cursor, dy_cursor))
                speed_px_s = displacement / max(dt, 1e-6)

                locked = stationary_lock.update(speed_px_s, dt)

                if not locked:
                    MAX_SPEED_PX_S = 5200.0
                    max_step = MAX_SPEED_PX_S * dt
                    if displacement > max_step:
                        scale = max_step / displacement
                        target_x = (current_cursor_x + dx_cursor * scale)
                        target_y = (current_cursor_y + dy_cursor * scale)

                    micro_deadzone = max(adaptive_deadzone, 2.2)
                    dx_final = target_x - current_cursor_x
                    dy_final = target_y - current_cursor_y

                    if np.hypot(dx_final, dy_final) >= micro_deadzone:
                        current_cursor_x = float(target_x)
                        current_cursor_y = float(target_y)
                        pyautogui.moveTo(
                            int(round(current_cursor_x)),
                            int(round(current_cursor_y)),
                            duration=0

                        locked_hand_y = float(cursor_tip_y)

            current_mode,
            is_paused,
            stationary_lock.locked if current_mode == "CURSOR" else False
        )

        tip_px = (int(hybrid_x * w), int(hybrid_y * h))
        palm_px = (int(smooth_palm_x * w), int(smooth_palm_y * h))

                frame,

                (10, h - 18),
                cv2.FONT_HERSHEY_SIMPLEX,
                0.48,
                (255, 255, 0),
